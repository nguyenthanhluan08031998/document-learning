Tự viết một logger bằng TypeScript

Thực tế nên dùng pino. Nhưng tự viết một logger nhỏ là bài tập rất tốt để thấy OOP và design pattern phối hợp ra sao: interface, Strategy cho nơi ghi, child logger, redaction, và AsyncLocalStorage cho ngữ cảnh request.

01Thiết kế

  • Logger nhận lời gọi info, error…, lọc theo level, gộp ngữ cảnh, che dữ liệu nhạy cảm, tạo một LogEntry.
  • Transport quyết định entry đi đâu: stdout dạng JSON, console có màu, bộ nhớ (cho test), HTTP. Mỗi transport hiện thực cùng một interface.
  • child() tạo logger mới dùng chung transport nhưng có thêm trường cố định.

02Kiểu dữ liệu & level

// logger/types.ts
export const LEVELS = { trace: 10, debug: 20, info: 30, warn: 40, error: 50, fatal: 60 } as const;
export type Level = keyof typeof LEVELS;

export type Fields = Record<string, unknown>;

export interface LogEntry {
  time: string;
  level: Level;
  msg: string;
  fields: Fields;
}

export interface Transport {
  write(entry: LogEntry): void;
}

03Transport: mỗi nơi ghi một lớp

// logger/transports.ts

// Production: one JSON object per line on stdout
export class JsonStdoutTransport implements Transport {
  write(e: LogEntry) {
    process.stdout.write(JSON.stringify({ time: e.time, level: e.level, msg: e.msg, ...e.fields }) + "\n");
  }
}

// Development: readable, coloured output
export class PrettyConsoleTransport implements Transport {
  private colors: Record<Level, string> = {
    trace: "\x1b[90m", debug: "\x1b[36m", info: "\x1b[32m",
    warn: "\x1b[33m", error: "\x1b[31m", fatal: "\x1b[35m",
  };
  write(e: LogEntry) {
    const extra = Object.keys(e.fields).length ? " " + JSON.stringify(e.fields) : "";
    console.log(`${e.time.slice(11, 23)} ${this.colors[e.level]}${e.level.toUpperCase()}\x1b[0m ${e.msg}${extra}`);
  }
}

// Tests: keep entries in memory to assert on them
export class MemoryTransport implements Transport {
  readonly entries: LogEntry[] = [];
  write(e: LogEntry) { this.entries.push(e); }
}

// Send to several destinations at once (Composite)
export class MultiTransport implements Transport {
  constructor(private targets: Transport[]) {}
  write(e: LogEntry) { this.targets.forEach(t => t.write(e)); }
}

04Lớp Logger

// logger/logger.ts
export interface LoggerOptions {
  level: Level;
  transport: Transport;
  base?: Fields;
  redactKeys?: string[];
}

export class Logger {
  private readonly minLevel: number;
  private readonly redactKeys: Set<string>;

  constructor(private readonly opts: LoggerOptions) {
    this.minLevel = LEVELS[opts.level];
    this.redactKeys = new Set((opts.redactKeys ?? ["password", "token", "authorization"]).map(k => k.toLowerCase()));
  }

  trace(msg: string, fields?: Fields) { this.log("trace", msg, fields); }
  debug(msg: string, fields?: Fields) { this.log("debug", msg, fields); }
  info(msg: string, fields?: Fields)  { this.log("info", msg, fields); }
  warn(msg: string, fields?: Fields)  { this.log("warn", msg, fields); }
  error(msg: string, fields?: Fields) { this.log("error", msg, fields); }
  fatal(msg: string, fields?: Fields) { this.log("fatal", msg, fields); }

  isEnabled(level: Level) { return LEVELS[level] >= this.minLevel; }

  // New logger sharing the transport, with extra fixed fields
  child(fields: Fields): Logger {
    return new Logger({ ...this.opts, base: { ...this.opts.base, ...fields } });
  }

  private log(level: Level, msg: string, fields: Fields = {}) {
    if (!this.isEnabled(level)) return;                      // cheap early exit
    const merged = { ...this.opts.base, ...fields };
    this.opts.transport.write({
      time: new Date().toISOString(),
      level,
      msg,
      fields: this.sanitize(merged) as Fields,
    });
  }

  // Redact secrets and turn Error objects into plain data
  private sanitize(value: unknown, depth = 0): unknown {
    if (depth > 5) return "[Too deep]";
    if (value instanceof Error) {
      return { name: value.name, message: value.message, stack: value.stack,
               cause: value.cause ? this.sanitize(value.cause, depth + 1) : undefined };
    }
    if (Array.isArray(value)) return value.map(v => this.sanitize(v, depth + 1));
    if (value && typeof value === "object") {
      return Object.fromEntries(Object.entries(value).map(([k, v]) =>
        [k, this.redactKeys.has(k.toLowerCase()) ? "[REDACTED]" : this.sanitize(v, depth + 1)]));
    }
    return value;
  }
}
Chữ ký ở đây là info(msg, fields) cho dễ đọc. pino dùng thứ tự ngược lại info(fields, msg); chọn một kiểu và giữ nhất quán.

05Ngữ cảnh theo request

// logger/context.ts
import { AsyncLocalStorage } from "node:async_hooks";

const storage = new AsyncLocalStorage<Logger>();

export function runWithLogger<T>(logger: Logger, fn: () => T): T {
  return storage.run(logger, fn);
}

// Returns the request-scoped logger if we are inside a request, otherwise the root
export function currentLogger(root: Logger): Logger {
  return storage.getStore() ?? root;
}

06Lắp ráp & sử dụng

// main.ts — composition root decides the transport per environment
const isDev = process.env.NODE_ENV === "development";

export const rootLogger = new Logger({
  level: (process.env.LOG_LEVEL as Level) ?? (isDev ? "debug" : "info"),
  transport: isDev ? new PrettyConsoleTransport() : new JsonStdoutTransport(),
  base: { service: "order-svc", version: process.env.APP_VERSION },
});
export const log = () => currentLogger(rootLogger);

// Express middleware: one child logger per request
app.use((req, res, next) => {
  const reqId = req.header("x-request-id") ?? crypto.randomUUID();
  const reqLog = rootLogger.child({ reqId, path: req.path });
  const start = Date.now();
  res.on("finish", () => reqLog.info("request completed", { status: res.statusCode, ms: Date.now() - start }));
  runWithLogger(reqLog, next);
});

// Deep inside business code
export async function placeOrder(input: PlaceOrderInput) {
  log().info("placing order", { userId: input.userId, items: input.items.length });
  try {
    await payment.charge(input);
  } catch (err) {
    log().error("payment failed", { err, password: "will be redacted" });
    throw err;
  }
}

07Test

import { test, expect } from "vitest";

test("filters by level, merges child fields and redacts secrets", () => {
  const memory = new MemoryTransport();
  const logger = new Logger({ level: "info", transport: memory, base: { service: "test" } });

  logger.debug("hidden");                                    // below min level
  logger.child({ reqId: "r1" }).info("login", { user: "an", password: "123" });

  expect(memory.entries).toHaveLength(1);
  expect(memory.entries[0].fields).toEqual({ service: "test", reqId: "r1", user: "an", password: "[REDACTED]" });
});
